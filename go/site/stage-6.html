第一课 · net/http 裸写 讲义就绪 · 未开课

实战热身：不用框架写一个 HTTP 服务，知道 gin 帮你省了什么。练习并入 code/13-gin/。

package main

import (
    "encoding/json"
    "net/http"
)

type Device struct {
    Name  string  `json:"name"`
    Price float64 `json:"price"`
}

var devices = []Device{{"AP1", 99}, {"AP2", 199}}   // 内存版「数据库」

func main() {
    http.HandleFunc("/api/devices", func(w http.ResponseWriter, r *http.Request) {
        switch r.Method {                            // 手动分发方法
        case http.MethodGet:
            w.Header().Set("Content-Type", "application/json")   // 手动设头
            json.NewEncoder(w).Encode(devices)       // 手动序列化写响应
        case http.MethodPost:
            var d Device
            if err := json.NewDecoder(r.Body).Decode(&d); err != nil {   // 手动解 body
                http.Error(w, "bad json", http.StatusBadRequest)
                return
            }
            devices = append(devices, d)             // 阶段 2 append 知识实战
            w.WriteHeader(http.StatusCreated)
        default:
            http.Error(w, "method not allowed", http.StatusMethodNotAllowed)
        }
    })
    http.ListenAndServe(":8080", nil)   // :8080 监听；nil = 用默认路由表
}
# 起服务后自己开另一个终端验证（环境命令自己敲）：
curl http://localhost:8080/api/devices
curl -X POST http://localhost:8080/api/devices -d "{\"name\":\"AP3\",\"price\":50}"
curl http://localhost:8080/api/devices     # 第三条出现了
JS 对照

≈ node 原生 http.createServer + 手动路由——没框架时 Node 也这么苦。痛点同步显现：路由参数要手切 URL、中间件要手串、没有统一 JSON 响应。这些痛点就是 gin 存在的意义，下节课逐一对照。

back 对照 · 一切框架的地基

标准库的 http.Handler 接口只有 ServeHTTP(w, r) 一个方法——gin 的整个 engine 实现的就是它（阶段 3 接口的终极实战：gin 不是取代 net/http，是实现它的 Handler）。http.HandlerFunc 更妙：func(w, r) 类型的函数自动实现 Handler 接口——函数是一等公民 + 隐式接口的组合拳，你传的所有匿名 handler 都靠它。

第二课 · gin 框架 讲义就绪 · 未开课

练习落在 code/13-gin/。net_scan/smb 的骨架。

1. 路由与参数（完整可跑版）

package main

import "github.com/gin-gonic/gin"

func main() {
    r := gin.Default()                      // 自带 Logger + Recovery 中间件

    api := r.Group("/api/v1")               // 路由组 ≈ express.Router() 挂载
    {
        api.GET("/devices", listDevices)            // 查询参数 ?q=
        api.GET("/devices/:id", getDevice)          // 路径参数 :id
        api.POST("/devices", createDevice)          // body JSON
    }
    r.Run(":8080")                          // ≈ app.listen(8080)
}

func listDevices(g *gin.Context) {
    q := g.Query("q")                       // ?q=ap（缺省空串）
    g.JSON(200, gin.H{"q": q, "items": []string{"AP1", "AP2"}})   // gin.H ≈ 直接 res.json({...})
}

func getDevice(g *gin.Context) {
    id := g.Param("id")                     // /devices/ap-1 → id = "ap-1"
    g.JSON(200, gin.H{"id": id})
}

func createDevice(g *gin.Context) {
    var req struct {                        // 请求 DTO：匿名 struct 就地定义
        Name  string `json:"name" binding:"required"`    // binding tag = 校验规则
        Price int    `json:"price"`
    }
    if err := g.BindJSON(&req); err != nil {           // body → struct；校验失败自动 400
        return                              // BindJSON 失败时响应已写好，直接 return
    }
    g.JSON(201, gin.H{"created": req.Name})
}
curl http://localhost:8080/api/devices?q=ap
curl http://localhost:8080/api/devices/ap-1
curl -X POST http://localhost:8080/api/devices -d '{"name":"AP3","price":50}'
curl -X POST http://localhost:8080/api/devices -d '{"price":50}'    # 缺 name → 自动 400

2. 中间件

r.Use(func(c *gin.Context) {          // r.Use = 全局；group.Use = 组内
    start := time.Now()
    c.Next()                            // 放行给后续 handler（先序段到此为止）
    log.Println(time.Since(start))      // 回来记耗时（后序段——响应已写完）
})

3. JS 对照

JS 对照（Express 老家）

RouterGroup ≈ express.Router()；g.Param("id") ≈ req.params.id；g.Query ≈ req.query；BindJSON ≈ body-parser + 校验二合一（校验规则写在 struct tag：binding:"required"，不用 Joi 手写 schema）；c.Next() 中间件链 ≈ next()（gin 是先序+后序两段，不是真洋葱）；gin.H ≈ 直接 res.json({...})。

易踩坑：handler 忘 return 是经典 bug——c.JSON 只是写响应不中断函数，后面再写一个 c.JSON 就 double write（back 源码每个错误分支都紧跟 return 的原因，阶段 3 错误处理三段式在 gin 里的形态）。

4. back 真实代码

back 真实代码 ×2

net_scan 入口（cmd/server/main.go）——三行骨架 + 中间件实战：

r := gin.Default()
r.Use(func(c *gin.Context) {                    // CORS 中间件（真实生产代码）
    c.Writer.Header().Set("Access-Control-Allow-Origin", "*")
    ...
    c.Next()
})
api.RegisterRoutes(r)                           // 路由注册收敛到一个函数

权限路由组（internal/api/handlers.go）——路由组 + 中间件组合出权限层级：

authAware := authGroup(api)                    // 登录就能访问
{
    authAware.GET("/auth/me", GetAuthMe)
    authAware.POST("/logout", requireAuthenticated(), Logout)   // 路由级中间件
}
scanRoutes := permissionGroup(api, func(perms authPermissions) bool {   // 按权限位建组
    return perms.CanScan                        // 有扫描权限才放行
})
{
    scanRoutes.POST("/scan", TriggerScan)
}

handler 本体是函数式的（结构最简，阶段 1 就见过）：GetDevices(c *gin.Context) 里 Bind/Query/JSON 三板斧 + 错误分支 return。smb 的路由注册则是组 → controller 循环（接口 + 隐式实现 + map 的合体，阶段 3 见过的 RegisterRoutes）。

第三课 · gorm ORM 讲义就绪 · 未开课

练习落在 code/14-gorm/（sqlite 免装数据库）。

1. 模型：struct + 双 tag（阶段 2 知识的正式兑现）

type Device struct {
    ID    uint   `gorm:"primaryKey"`
    Name  string `gorm:"type:varchar(64);uniqueIndex"`   // 唯一索引——插重复会报错
    Price int    `gorm:"default:0"`
    CreatedAt int64 `gorm:"autoCreateTime:milli"`        // smb Site 同款（阶段 2 读过的那个字段）
}
func (Device) TableName() string { return "devices" }    // smb entity 同款方法（阶段 3 值接收者）

阶段 2 精读 smb entity/site.go 时认识的 gorm tag，这里是亲手用：primaryKey / type / default / uniqueIndex / autoCreateTime:milli / column——同一套标签体系，从那一刻起你就已经会读 smb 的实体定义了。

2. 连接 + 建表 + CRUD 四式（完整版）

db, err := gorm.Open(sqlite.Open("dev.db"), &gorm.Config{})
if err != nil {
    log.Fatal(err)                          // 连库失败没有活路
}
db.AutoMigrate(&Device{})                   // 按 struct + gorm tag 建表（≈ 迁移脚本自动生成）

// INSERT（返回值都在 *Device 上回填：ID 自增主键写回 d.ID）
d := Device{Name: "AP1", Price: 99}
if err := db.Create(&d).Error; err != nil {  // .Error 取错误——gorm 的链式调用统一这个姿势
    fmt.Println("插入失败:", err)              // 比如违反唯一索引
}

// SELECT 单条（? 占位防注入——字符串拼接 SQL 是安全红线）
var found Device
db.First(&found, "name = ?", "AP1")          // 找不到 → ErrRecordNotFound（errors.Is 判它）

// SELECT 列表 + 条件
var list []Device
db.Where("price > ?", 50).Find(&list)

// UPDATE / DELETE
db.Model(&found).Update("price", 79)
db.Delete(&found)

三个要点：① gorm 的链式调用结果都有 .Error 字段——阶段 3 的 if err != nil 在 ORM 层的形态；② First 找不到记录返回 gorm.ErrRecordNotFound，用 errors.Is(err, gorm.ErrRecordNotFound) 判定（错误链实战）；③ db.Debug() 打印真实 SQL（如 db.Debug().Where(...).Find(...)）——复杂查询看真身是排障标配。

3. JS 对照

JS 对照

gorm ≈ Sequelize/TypeORM：模型定义（struct+tag ≈ Model+装饰器）、关联、迁移。坑也同款：复杂查询生成的 SQL 不直观，要开日志看真身（db.Debug() ≈ Sequelize logging）。? 占位符 ≈ 参数化查询，字符串拼接 SQL 在两边都是安全红线。差异：gorm 无 eager/lazy loading 之分，关联要显式 Preload；错误不抛异常走 .Error 字段。

4. back 对照

back 真实代码

smb 不直接用 gorm 裸 API，而是套了 gobase 泛型仓储（Insert/FindByWhere...），底层还是 gorm——阶段 7 精读拆洋葱。net_scan 更直接：storage 包里就是 gorm 标准用法（repo_mysql.go 里 json.Marshal 切片存 TEXT 列那个文件——阶段 2 见过的模式），先读它。

第四课 · 分层架构与终极练习 讲义就绪 · 未开课

练习落在 code/15-gin-crud/——阶段 6 毕业作品。

1. back 的分层地图

cmd/boot.go        入口：组装依赖、启动
api/restserver.go  路由注册：gin 分组 → 各 controller
controller/        参数解析、校验、调 service、统一响应
service/           业务逻辑、事务、DTO 定义
repo/              数据访问（smb: 泛型仓储 / net_scan: storage 包）
model/entity/      struct + 双 tag（阶段 2 精读过的 Site）

一次请求的一生：HTTP → 路由匹配 → controller 解参数 → service 干业务 → repo 读写库 → 原路返回 JSON。每层单向依赖，entity 被所有人引用。

JS 对照

≈ 前端分层心智：routes → pages → api/services → store——只是方向从「渲染」换成「数据处理」。
